### 🤖 תגובה 9: להלן סקריפט פייתון (Python) מלא המיועד לרוץ בתיקייה שבה שמורים קובצי ה-HTML…

להלן סקריפט פייתון (Python) מלא המיועד לרוץ בתיקייה שבה שמורים קובצי ה-HTML שלכם. 

הסקריפט סורק את התיקייה, פותח כל קובץ באופן בטוח, מחלץ מתוכו את תגית ה-`<title>` הפנימית, ומחליף את שם הקובץ לשם הכותרת שמצא.

### תכונות מובנות בסקריפט:
* **תמיכה מלאה בעברית** (קידוד UTF-8 ופענוח ישויות HTML כגון גרשיים או מירכאות).
* **ניקוי תווים לא חוקיים** (מנקה אוטומטית תווים שאסור להשתמש בהם בשמות קבצים במערכות ההפעלה, כמו `\`, `/`, `:`, `*`, `?`, `"`, `<`, `>`, `|` ומחליף אותם בקו תחתון).
* **מניעת דריסת קבצים** (במידה וישנם שני קבצים בעלי כותרת זהה, הסקריפט יוסיף מספור אוטומטי בסוגריים, לדוגמה: `(1)` או `(2)`, כדי למנוע מחיקת מידע).

### קוד הסקריפט (`rename_html.py`):

```python
import os
import re
import glob
import html

def sanitize_filename(name):
    """
    מנקה את שם הקובץ מתווים שאינם חוקיים במערכות הפעלה שונות
    """
    # החלפת תווים אסורים בקו תחתון
    sanitized = re.sub(r'[\/\\:\*\?"<>\|]', '_', name)
    # ניקוי רווחים כפולים או ירידות שורה מיותרות
    sanitized = re.sub(r'\s+', ' ', sanitized).strip()
    return sanitized

def rename_html_files_to_title(directory="."):
    """
    סורק את התיקייה ומחליף את שמות קבצי ה-HTML לכותרת הפנימית שלהם
    """
    # מציאת כל קבצי ה-HTML בתיקייה שצוינה
    html_files = glob.glob(os.path.join(directory, "*.html")) + glob.glob(os.path.join(directory, "*.htm"))
    
    if not html_files:
        print("לא נמצאו קבצי HTML בתיקייה שצוינה.")
        return

    print(f"נמצאו {len(html_files)} קבצים לעיבוד.\n")

    for filepath in html_files:
        filename = os.path.basename(filepath)
        
        # ניסיון קריאת הקובץ בקידודים שונים (תמיכה מלאה בעברית)
        content = ""
        try:
            with open(filepath, 'r', encoding='utf-8') as f:
                content = f.read()
        except UnicodeDecodeError:
            try:
                # ניסיון קידוד חלופי במידה ו-UTF-8 נכשל (Windows-1255 הנפוץ בעברית)
                with open(filepath, 'r', encoding='cp1255', errors='ignore') as f:
                    content = f.read()
            except Exception as e:
                print(f"❌ שגיאה בקריאת הקובץ {filename}: {e}")
                continue
        except Exception as e:
            print(f"❌ שגיאה בקריאת הקובץ {filename}: {e}")
            continue

        # חילוץ תוכן תגית ה-title באמצעות Regex בצורה גמישה
        title_match = re.search(r'<title[^>]*>(.*?)</title>', content, re.IGNORECASE | re.DOTALL)
        
        if title_match:
            raw_title = title_match.group(1)
            # המרת ישויות HTML (כגון &quot; או &#39; לתווים רגילים)
            raw_title = html.unescape(raw_title)
            
            # ניקוי שם הקובץ החדש
            new_name = sanitize_filename(raw_title)
            
            if not new_name:
                print(f"⚠️ דילוג על {filename}: הכותרת שנמצאה אינה חוקית כשם קובץ.")
                continue
            
            # בניית נתיב היעד החדש
            new_filepath = os.path.join(directory, f"{new_name}.html")
            
            # מנגנון מניעת דריסה במקרה של כותרות זהות
            counter = 1
            while os.path.exists(new_filepath):
                # אם הקובץ הקיים הוא בדיוק הקובץ הנוכחי, אין צורך לשנות שם
                if os.path.abspath(filepath) == os.path.abspath(new_filepath):
                    break
                new_filepath = os.path.join(directory, f"{new_name} ({counter}).html")
                counter += 1
            
            # בדיקה האם השם כבר תואם לכותרת
            if os.path.abspath(filepath) == os.path.abspath(new_filepath):
                print(f"ℹ️ הקובץ {filename} כבר נושא את שם הכותרת שלו.")
                continue

            try:
                os.rename(filepath, new_filepath)
                print(f"✅ שונה בהצלחה: {filename} ➔ {os.path.basename(new_filepath)}")
            except Exception as e:
                print(f"❌ שגיאה בשינוי השם של {filename}: {e}")
        else:
            print(f"⚠️ דילוג על {filename}: לא נמצאה תגית <title> פנימית בקוד.")

if __name__ == "__main__":
    # כברירת מחדל, הסקריפט ירוץ על התיקייה שבה הוא ממוקם
    rename_html_files_to_title()
```
